# IR保留扩展构建入口；依赖原Icraft工具链，不会自动执行板端推理。
cmake_minimum_required(VERSION 3.16)
project(IR_probe_m834 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Icraft-HostBackend REQUIRED)
if(WIN32)
  add_executable(IR_backend_trace trace.cpp)
  target_compile_options(IR_backend_trace PRIVATE /utf-8 /Zc:preprocessor /Zc:__cplusplus)
  target_link_libraries(IR_backend_trace PRIVATE Icraft::HostBackend)
else()
  if(NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "aarch64")
    message(FATAL_ERROR "Use the existing frozen AArch64 toolchain")
  endif()
  find_package(icraft-zg330backend REQUIRED)
  add_executable(IR_NPU_probe probe.cpp)
  target_link_options(IR_NPU_probe PRIVATE "-Wl,-rpath-link,/usr/lib/aarch64-linux-gnu")
  target_link_libraries(IR_NPU_probe PRIVATE Icraft::HostBackend Icraft::ZG330Backend)
endif()
