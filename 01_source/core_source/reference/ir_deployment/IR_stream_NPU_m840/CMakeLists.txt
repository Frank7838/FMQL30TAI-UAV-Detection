# IR保留扩展构建入口；依赖原Icraft工具链，不会自动执行板端推理。
cmake_minimum_required(VERSION 3.16)
project(IR_stream_NPU_m840 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(WIN32)
 add_executable(IR_decode_replay replay.cpp)
 target_compile_options(IR_decode_replay PRIVATE /utf-8)
else()
 if(NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "aarch64")
  message(FATAL_ERROR "Use existing AArch64 toolchain")
 endif()
 find_package(Icraft-HostBackend REQUIRED)
 find_package(icraft-zg330backend REQUIRED)
 add_executable(IR_NPU_stream stream.cpp)
 target_link_options(IR_NPU_stream PRIVATE "-Wl,-rpath-link,/usr/lib/aarch64-linux-gnu")
 target_link_libraries(IR_NPU_stream PRIVATE Icraft::HostBackend Icraft::ZG330Backend)
endif()
