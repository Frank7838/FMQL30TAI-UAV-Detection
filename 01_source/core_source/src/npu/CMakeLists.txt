cmake_minimum_required(VERSION 3.16)
project(drone_m91_arm64 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "aarch64")
  message(FATAL_ERROR "Use the existing aarch64 toolchain")
endif()
list(APPEND CMAKE_PREFIX_PATH /usr/cmake)
find_package(Icraft-BuyiBackend REQUIRED)
find_package(Icraft-HostBackend REQUIRED)
find_package(icraft-zg330backend REQUIRED)
find_package(OpenCV REQUIRED PATHS /ModelzooDeps/aarch64/Static/lib/cmake/opencv4 NO_DEFAULT_PATH)
find_package(yaml-cpp REQUIRED PATHS /ModelzooDeps/aarch64/Static/lib/cmake/yaml-cpp NO_DEFAULT_PATH)
# Submission layout adapter only; SDK packages still come from the pinned toolchain.
set(RUNTIME_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/icraft_runtime" CACHE PATH "Vendor source dependency")
add_executable(product_stream product_stream.cpp)
foreach(app product_stream )
  target_include_directories(${app} PRIVATE "${RUNTIME_ROOT}/src" "${RUNTIME_ROOT}/modelzoo_utils"
    /ModelzooDeps/include/Eigen-3.4.0 ${OpenCV_INCLUDE_DIRS})
  target_link_directories(${app} PRIVATE /ModelzooDeps/aarch64/Static/lib)
  target_link_options(${app} PRIVATE "-Wl,-rpath-link,/usr/lib/aarch64-linux-gnu")
  target_link_libraries(${app} PRIVATE Icraft::HostBackend Icraft::BuyiBackend Icraft::ZG330Backend ${OpenCV_LIBS} yaml-cpp::yaml-cpp)
endforeach()
